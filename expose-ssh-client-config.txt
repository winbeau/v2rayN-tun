# ============================================================
#  连 brivanMac 的 SSH 配置
#  把下面这段追加到你自己电脑的 ~/.ssh/config，然后 ssh mac
# ============================================================

Host mac
    HostName 1.92.76.152
    Port 22001
    User winbeau
    IdentityFile ~/.ssh/id_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 30
    ServerAliveCountMax 3
    StrictHostKeyChecking accept-new

# ============================================================
#  不改 config 的话，一条命令也行：
#
#    ssh -p 22001 winbeau@1.92.76.152
#
# ============================================================
#  前提
#
#  1. 你的公钥要在对方机器的 ~/.ssh/authorized_keys 里
#     把你的 ~/.ssh/id_ed25519.pub 发过去
#
#  2. 对方要先开隧道：./scripts/expose-ssh.sh on
#     关掉之后端口立刻消失，连不上是正常的
#
# ============================================================
#  说明
#
#  · 只认密钥登录，密码不行
#  · 换了网络后隧道会在 3 秒内自动重连，不用对方重开
#  · 连不通先看对方有没有开隧道，再看自己的公钥有没有被授权
# ============================================================
