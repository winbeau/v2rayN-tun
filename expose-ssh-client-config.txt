# ============================================================
#  连 brivanMac 的 SSH 配置
#  用法：把下面这段追加到你自己电脑的 ~/.ssh/config
#        然后    ssh mac
# ============================================================

Host mac
    HostName 127.0.0.1
    ProxyJump winbeau@124.71.228.242
    Port 30022
    User winbeau
    IdentityFile ~/.ssh/id_ed25519
    IdentitiesOnly yes
    ServerAliveInterval 30
    ServerAliveCountMax 3
    StrictHostKeyChecking accept-new

# ============================================================
#  不想改 config 的话，用这一条命令：
#
#    ssh -J winbeau@124.71.228.242 -p 30022 winbeau@127.0.0.1
#
# ============================================================
#  前提（必须先满足）
#
#  1. 你的公钥要被加到对方机器的 ~/.ssh/authorized_keys
#     把你的 ~/.ssh/id_ed25519.pub 发过去
#
#  2. 对方要先开隧道：
#       EXPOSE_JUMP=huawei2 ./scripts/expose-ssh.sh on
#     关掉之后（off）端口会立刻消失，连不上是正常的
#
# ============================================================
#  说明
#
#  · 目标写 127.0.0.1 是因为反向端口绑在跳板机的回环地址上，
#    这个 127.0.0.1 是在【跳板机上】解析的，不是你的本机
#  · 只认密钥登录，密码不行
#  · 连不通先看对方有没有开隧道，再看自己的公钥有没有被授权
#  · 如果 124.71.228.242 不通，可以换 1.92.76.152（同一套镜像）
# ============================================================
